## 实验平台

1. Ubuntu18.04 (Windows11 + WSL2)
2. gcc 7.5.0

## 实验要求

实现ping，该命令行程序使用方法为：

```bash
sudo ping <目标ip> -l <报文长度> -n <发送次数>
```

> Linux平台下必须使用sudo权限，否则无法使用ping

本实验中，`l`默认为64，`n`默认为4

---

## 基本原理

实现ping，其实就是把下面的三个步骤实现`n`次：

1. 建立socket连接
2. 发送报文
3. 接受报文

每次接受报文后，解包后，将响应IP、Seq、报文大小、`RTT`和`TTL`打印在屏幕上。循环结束后，将总的信息统计打印在屏幕上（发送了几次，成功接受了几次，然后算出成功接受率`accept`和丢包率`lost`）

### 建立socket连接

Linux下使用C语言，可以直接使用Linux提供的库来建立一个基于ICMP协议的socket连接

```c
struct protoent *protocol = getprotobyname("icmp");
int sockfd = socket(AF_INET, SOCK_RAW, protocol->p_proto);
```

我们只需要设置发送的报文的报文头，缓冲区即可。

建立后通过`sendto`函数发送缓冲区内容，通过`recvfrom`函数阻塞地接受缓冲区内容。

所有请求结束后，通过`close(ping.sockfd);`来断开socket连接。

### 发送ICMP报文

通过`sendto`函数来发送ICMP报文，具体的函数调用和各个变量含义如下：

```c
int res = sendto(
	ping.sockfd, 			// 建立的socket连接的标识ID
	ping.send_buffer,		// 发送报文内容
	icmp_size,				// ICMP报文在缓冲区中对应的长度
	0,						// flag位，置为0即可
	dst_addr,				// 目标ip地址
	dst_addr_size			// 目标ip地址长度
);
```

返回值为-1则失败，否则成功。

所以发送ICMP报文的关键在于如何设置报文内容，也就是`ping.send_buffer`。在实验中，我们这么设计：

```c
icmp_pack->icmp_type = ICMP_ECHO;
icmp_pack->icmp_cksum = 0;
icmp_pack->icmp_code = 0;
icmp_pack->icmp_id = ping.pid;
icmp_pack->icmp_seq = pack_id;
icmp_pack->icmp_cksum = check_sum((unsigned short *)icmp_pack, icmp_pack_length);
```

具体的细节可以看后续的实现代码

### 接收ICMP报文

通过`recvfrom`函数来接受ICMP报文，具体调用如下：

```c
int res = recvfrom(
	ping.sockfd,				// 建立的socket连接的标识ID
	ping.recv_buffer,			// 接收报文内容
	sizeof(ping.recv_buffer),	// 接收报文长度
	0,							// flag位
	src_addr,					// 原ip地址
	&src_addr_len				// 原ip地址长度的指针（我也不知道为什么这个位置参数设计成socklen_t *）
);
```

`res`的含义同发送报文的返回值。

所有的全局变量都封装在结构体`ping`中，比如想要全局访问当前socket连接的ID可以通过`ping.socketfd`来访问。`ping`的定义如下：

```c
struct Ping {   
    // global buffer
    char send_buffer[4096], recv_buffer[4096];
    // address
    struct sockaddr_in dst_addr, src_addr;
    // receive number and receive number
    size_t send_num, recv_num;
    // data length
    size_t data_length, sockfd;
    // pid 
    pid_t pid;
    // st
    struct timeval recv_st, send_st;
    // failed time
    size_t fail_time;
    // timeout
    size_t timeout;
} ping;
```

为了正确接受参数，我还设置了一个结构体来存储和解析命令行参数，这不是重点，所以只给出定义，解析的实现就不给出了：

```c
// parse arguments
struct Args {
    char dst_addr[64]; 
    size_t data_length, counts;    
};

// parse arguments in command prompt
struct Args parse_args(int argc, char const *args[]);
```

### 入口函数

入口函数的实现如下：

```c
int main(int argc, char const *argv[]) {
    if (argc < 2) {
        printf("usage %s <target address> -l <data length> -n <counts>\n", argv[0]);
        exit(-1);
    }
    struct Args args = parse_args(argc, argv);
                                     
    // init
    ping.data_length = args.data_length;
    ping.send_num = ping.recv_num = 0;
    
    // welcome
    printf("%s\n", welcome);

    struct protoent *protocol = getprotobyname("icmp");
    ping.sockfd = socket(AF_INET, SOCK_RAW, protocol->p_proto);
    
    ping.dst_addr.sin_family = AF_INET;
    ping.src_addr.sin_family = AF_INET;
    ping.fail_time = 0;

    // transform ip to host
    struct hostent *host = gethostbyname(args.dst_addr);        // acquire host name
    if (host == NULL) {
        printf("[ERROR] Nonexist Host name!\n");
        exit(-1);
    }
    memcpy((char *)&ping.dst_addr.sin_addr, host->h_addr_list[0], host->h_length);
    
    // set priviledge of current account
    setuid(getuid());
    ping.pid = getpid();

    // set timeout when receive icmp
    ping.timeout = 5;

    // repeat counts times
    for (int i = 0; i < args.counts; ++ i) {
        // send icmp pack
        send_icmp();
        // recevie icmp pack
        recv_icmp();
        sleep(1);
    }

    // finish and report
    float recv_rate = (float)ping.recv_num / (float)ping.send_num * 100;
    printf("[PING] %ld sended, %ld received, accept %.0f%%, lost %.0f%%\n", 
        ping.send_num, 
        ping.recv_num,
        recv_rate,
        100 - recv_rate
    );
    close(ping.sockfd);

    return 0;
}
```

很明显，最核心的两个函数就是`send_icmp()`和`recv_icmp()`。

### send_icmp

为了实现`send_icmp`，需要实现两个函数：

1. 计算校验和
2. 封装icmp报文

由于之前已经介绍过，此处直接给出代码：

#### 计算校验和

```c
unsigned short check_sum(unsigned short *addr, int length) {
    int sum = 0, i = length;
    unsigned short *p = addr;
    unsigned short ans = 0;
    while (i > 1) {
        sum += *(p ++);
        i -= 2;
    }
    if (i == 1) {
        *(unsigned char *)(&ans)=*(unsigned char *)p;
        sum += ans;
    }
    sum = (sum >> 16) + (sum & 0xffff);
    sum += (sum >> 16);
    ans = ~sum;
    return ans;
}
```

#### 封装报文头

```c
int set_icmp_head(int pack_id) {
    int i, icmp_pack_length = ping.data_length + 8;
    // init send content
    struct icmp *icmp_pack = (struct icmp *)ping.send_buffer;
    // use icmp pack format to fill send buffer
    icmp_pack->icmp_type = ICMP_ECHO;
    icmp_pack->icmp_cksum = 0;
    icmp_pack->icmp_code = 0;
    icmp_pack->icmp_id = ping.pid;
    icmp_pack->icmp_seq = pack_id;

    struct timeval *send_time = (struct timeval *)icmp_pack->icmp_data;
    // log send time
    gettimeofday(send_time, NULL);
    icmp_pack->icmp_cksum = check_sum((unsigned short *)icmp_pack, icmp_pack_length);
    return icmp_pack_length;
}
```

下面就是发送报文的函数了：

```c
void send_icmp() {
    int icmp_size;
    struct sockaddr* dst_addr = (struct sockaddr *) &ping.dst_addr;
    socklen_t dst_addr_size = sizeof(ping.dst_addr);
    
    ping.send_num ++;
    // get icmp head
    icmp_size = set_icmp_head(ping.send_num);
    // send icmp
    int res = sendto(
        ping.sockfd, 
        ping.send_buffer,
        icmp_size,
        0,
        dst_addr,
        dst_addr_size
    );

    if (res < 0) {
        printf("[PING] \033[31m Error to send ICMP! Errno %d\033[0m\n", errno);
    } 
}
```

### recv_icmp

接收ICMP报文分为接收和解析两个步骤，接收的函数如下：

```c

void recv_icmp() {
    struct sockaddr *src_addr = (struct sockaddr *)&ping.src_addr;
    socklen_t src_addr_len = sizeof(ping.src_addr);

    // prevent timeout
    alarm(ping.timeout);
    // attempt to receive
    int res = recvfrom(
        ping.sockfd,
        ping.recv_buffer,
        sizeof(ping.recv_buffer),
        0,
        src_addr,
        &src_addr_len
    );
    if (res < 0) {          // fail to recv the package
        printf("[PING] \033[31m Fail to receive ICMP! Errno %d\033[0m\n", errno);
        ping.fail_time ++;
        if (ping.fail_time > 5) {
            printf("[PING] \033[35mYou've failed too much time!\033[0m\n");
            exit(-1);
        }
    } else {
        gettimeofday(&ping.recv_st, NULL);
        if (parse_icmp_head(ping.recv_buffer, res) != -1)
            ping.recv_num ++;
    }
}
```

解析的步骤如下，我会在解析完成后直接把信息打印在屏幕上：

```c
int parse_icmp_head(char * ip_buffer, int length) {
    struct ip *ip_pack = (struct ip *)ip_buffer;
    // length of ip head is flag * 4
    int ip_head_len = ip_pack->ip_hl << 2;
    struct icmp *icmp_pack = (struct icmp *)(ip_buffer + ip_head_len);
    int icmp_len = length - ip_head_len;
    float RTT;

    if (icmp_len < 8) {     
        printf("[PING] \033[31m ICMP length is shorter than 8!\033[0m\n");
        return -1;
    }

    // ensure the sender is me
    if (icmp_pack->icmp_type == ICMP_ECHOREPLY && icmp_pack->icmp_id == ping.pid) {
        ping.send_st = *(struct timeval *)icmp_pack->icmp_data;
        timeval_sub(&ping.recv_st, &ping.send_st);
        RTT = (float)ping.recv_st.tv_sec * 1000. + (float)ping.recv_st.tv_usec / 1000.;
        printf("[PING] ICMP from %s: Seq=%ld  Byte=%d  RTT=%.2f ms  TTL=%d\n", 
            inet_ntoa(ping.src_addr.sin_addr),
            ping.send_num,
            icmp_len,
            RTT,
            ip_pack->ip_ttl
        );
    } else {
        return -1;
    }
}
```
